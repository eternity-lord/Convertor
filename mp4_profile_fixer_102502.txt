def find_boris_configs():
    paths = []
    bases = [r"C:\Program Files\BorisFX", r"C:\Program Files (x86)\BorisFX",
             os.path.expandvars(r"%ProgramData%\BorisFX"), os.path.expandvars(r"%AppData%\BorisFX")]
    for base in bases:
        try:
            for root, dirs, files in os.walk(base):
                for f in files:
                    if f.lower() == "s_config.text":
                        paths.append(Path(root) / f)
        except: pass
    return paths

def fix_boris_gpu_integration():
    configs = find_boris_configs()
    if not configs: return
    for cfg in configs:
        try:
            lines = cfg.read_text(encoding="utf-8", errors="ignore").splitlines()
            found = False
            new_lines = []
            for line in lines:
                if "host_gpu_integration" in line.lower() and ":" in line:
                    parts = line.split(":", 1)
                    if len(parts) == 2:
                        key = parts[0].strip()
                        value = parts[1].strip()
                        if value == "0":
                            line = f"{key}: 1"
                            log.info(f"Fixed: {cfg.name} -> {key}: 1")
                        else:
                            log.info(f"Already active: {key}: {value}")
                    found = True
                new_lines.append(line)
            if not found:
                new_lines.append("host_gpu_integration: 1")
                log.info(f"Added host_gpu_integration: 1 to {cfg.name}")
            cfg.write_text("\n".join(new_lines), encoding="utf-8")
        except Exception as e:
            log.error(f"Boris config error: {e}")

def find_power_plan(plan_name_hint):
    try:
        out = subprocess.run(["powercfg", "/list"], capture_output=True, text=True).stdout
        for line in out.splitlines():
            if plan_name_hint.lower() in line.lower():
                parts = line.split()
                for p in parts:
                    if p.count('-') == 4 and len(p) == 36: return p
    except: pass
    return None

def disable_visual_effects():
    if not HAVE_WIN32: return
    try:
        key = winreg.OpenKey(winreg.HKEY_CURRENT_USER, r"Software\Microsoft\Windows\CurrentVersion\Explorer\VisualEffects", 0, winreg.KEY_SET_VALUE)
        winreg.SetValueEx(key, "VisualFXSetting", 0, winreg.REG_DWORD, 2); winreg.CloseKey(key)
    except: pass

def get_edius_handle():
    for proc in psutil.process_iter(['pid', 'name']):
        if proc.info['name'].lower() == EDIUS_EXE.lower():
            pid = proc.info['pid']; handle = ctypes.windll.kernel32.OpenProcess(0x0200, False, pid)
            if handle: return pid, handle
    return None, None

def set_edius_priority():
    pid, handle = get_edius_handle()
    if not handle: return
    try:
        ctypes.windll.kernel32.SetPriorityClass(handle, HIGH_PRIORITY_CLASS)
        ioprio = ctypes.c_ulong(IO_PRIO_HIGH)
        ctypes.windll.kernel32.SetProcessInformation(handle, 33, ctypes.byref(ioprio), ctypes.sizeof(ioprio))
    except: pass
    finally: ctypes.windll.kernel32.CloseHandle(handle)

def kill_background_processes(custom_list=None):
    lst = custom_list or ["chrome.exe","firefox.exe","msedge.exe","onedrive.exe","spotify.exe","steam.exe","discord.exe"]
    for proc in psutil.process_iter(['pid','name']):
        try:
            if proc.info['name'].lower() in lst: proc.terminate()
        except: pass

def suspend_service(svc_name):
    global original_service_states
    if not HAVE_WIN32: return
    try:
        import win32serviceutil, win32service
        status = win32serviceutil.QueryServiceStatus(svc_name)[1]
        original_service_states[svc_name] = {'status': status}
        if status == win32service.SERVICE_RUNNING: win32serviceutil.StopService(svc_name)
        subprocess.run(["sc","config",svc_name,"start=","disabled"], capture_output=True)
    except: pass

def restore_services():
    global original_service_states
    if not HAVE_WIN32: return
    import win32serviceutil, win32service
    for svc, info in original_service_states.items():
        try:
            if info['status'] == win32service.SERVICE_RUNNING: win32serviceutil.StartService(svc)
            subprocess.run(["sc","config",svc,"start=","auto"], capture_output=True)
        except: pass
    original_service_states.clear()

def get_cpu_temp():
    if HAVE_WMI:
        try:
            w = wmi.WMI(namespace="root\\wmi"); temps = w.MSAcpi_ThermalZone()
            if temps: return temps[0].CurrentTemperature / 10.0 - 273.15
        except: pass
    try:
        for name, entries in psutil.sensors_temperatures().items():
            for e in entries:
                if 'core' in e.label.lower() or 'cpu' in e.label.lower(): return e.current
    except: pass
    return None

class HQXConverter(threading.Thread):
    def __init__(self, watch_folder, basic_settings, adv_settings, safe_mode=False):
        super().__init__(daemon=True)
        self.watch = Path(watch_folder)
        self.basic = basic_settings
        self.adv = adv_settings
        self.safe_mode = safe_mode
        self.processed = set()
        self._stop_event = threading.Event()

    def stop(self): self._stop_event.set()

    def run(self):
        log.info(f"Watching folder: {self.watch}")
        while not self._stop_event.is_set():
            try:
                for f in self.watch.iterdir():
                    if f.suffix.lower() in ['.avi','.mov','.mkv'] and f.name not in self.processed:
                        if self._is_stable(f):
                            self.processed.add(f.name)
                            self._convert(f)
            except Exception as e: log.error(f"Scan error: {e}")
            time.sleep(3)

    def _is_stable(self, filepath, wait=5):
        try:
            s1 = filepath.stat().st_size; time.sleep(wait)
            s2 = filepath.stat().st_size; time.sleep(1)
            return s1 == s2 == filepath.stat().st_size > 0
        except: return False

    def _convert(self, src):
        dst = src.with_suffix(".mp4")
        log.info(f"Converting: {src.name}")
        cmd = [FFMPEG_EXE, "-i", str(src), "-c:v", self.basic.get("video_codec", "h264_qsv")]
        cmd += ["-vf", "format=nv12"]
        cmd += ["-fflags", "+genpts","-vsync","vfr","-max_muxing_queue_size", "1024"]

        safe_keys = {"preset","profile","level","g","bf","look_ahead","look_ahead_depth",
                     "trellis","async_depth","rdo","scenario","extbrc","low_delay_brc",
                     "max_frame_size","max_frame_size_i","max_frame_size_p","max_frame_size_b",
                     "refs"}

        for key, val in self.adv.items():
            if val is None or val == "" or val is False: continue
            if self.safe_mode and key not in safe_keys: continue
            if key in ("adaptive_i","adaptive_b","forced_idr","mbbrc","fixed_framerate","low_power","repeat_pps","weight_pred","vcm"):
                continue
            flag = f"-{key}"
            if key == "profile": flag = "-profile:v"
            if isinstance(val, bool) and val: cmd += [flag, "1"]
            else: cmd += [flag, str(val)]

        rc = self.basic.get("rate_control", "VBR")
        if "global_quality" not in self.adv and rc == "CQP":
            cmd += ["-global_quality", str(self.basic.get("global_quality", 23))]
        if "b:v" not in self.adv and rc in ("VBR","CBR","ICQ","LA"):
            cmd += ["-b:v", self.basic.get("bitrate","15M")]
        if "maxrate" not in self.adv and rc == "VBR":
            cmd += ["-maxrate", self.basic.get("maxrate","20M")]
        if "bufsize" not in self.adv and rc == "VBR":
            cmd += ["-bufsize", self.basic.get("bufsize","30M")]

        if "preset" not in self.adv: cmd += ["-preset", self.basic.get("preset","veryslow")]
        if "profile" not in self.adv and self.basic.get("profile"): cmd += ["-profile:v", self.basic["profile"]]
        if "level" not in self.adv and self.basic.get("level"): cmd += ["-level", self.basic["level"]]
        gop = self.basic.get("gop_size",0)
        if "g" not in self.adv and gop > 0: cmd += ["-g", str(gop)]
        bf = self.basic.get("bframes", -1)
        if "bf" not in self.adv and bf >= 0: cmd += ["-bf", str(bf)]
        if "look_ahead" not in self.adv and self.basic.get("look_ahead", False):
            cmd += ["-look_ahead","1"]
            if "look_ahead_depth" not in self.adv: cmd += ["-look_ahead_depth", str(self.basic.get("look_ahead_depth",60))]

        cmd += ["-c:a", self.basic.get("audio_codec","aac"), "-b:a", self.basic.get("audio_bitrate","192k"), "-y", str(dst)]
        log.info("Running: " + " ".join(cmd))

        try:
            subprocess.run(cmd, check=True, capture_output=True, creationflags=subprocess.CREATE_NO_WINDOW)
            if dst.exists() and dst.stat().st_size > 0:
                log.info(f"Done: {dst.name}")
                if self.basic.get("delete_source", True): src.unlink(); log.info(f"Deleted {src.name}")
            else: log.error("Output file not created")
        except subprocess.CalledProcessError as e:
            log.error(f"FFmpeg error: {e.stderr.decode()}")

class MP4ProfileFixer(threading.Thread):
    """
    MP4 های خروجی EDIUS را می‌گیرد و فقط:
      1) Profile/Level را تغییر می‌دهد (مثلاً High 4.1)
      2) Bitrate را با VBR/CBR/CQP تنظیم می‌کند
      3) با QSV ری‌انکد می‌کند (سریع)
    بقیه تنظیمات فیلم (رزولوشن، فریم‌ریت، صدا) دست‌نخورده باقی می‌ماند.
    """
    def __init__(self, folder, profile, level, rc_mode, bitrate, maxrate, bufsize, cqp, delete_src=False):
        super().__init__(daemon=True)
        self.folder = Path(folder)
        self.profile = profile        # مثلاً "high"
        self.level = level            # مثلاً "4.1"
        self.rc_mode = rc_mode        # "VBR" / "CBR" / "CQP"
        self.bitrate = bitrate        # مثلاً "15M"
        self.maxrate = maxrate        # مثلاً "20M"
        self.bufsize = bufsize        # مثلاً "30M"
        self.cqp = cqp                # مثلاً "18"
        self.delete_src = delete_src
        self._stop_event = threading.Event()

    def stop(self):
        self._stop_event.set()

    def run(self):
        mp4_files = list(self.folder.glob("*.mp4"))
        # فایل‌هایی که قبلاً fix شدن رو رد کن
        mp4_files = [f for f in mp4_files if "_fixed" not in f.stem]

        if not mp4_files:
            log.info("MP4 Fixer: هیچ فایل MP4 پیدا نشد!")
            return

        log.info(f"MP4 Fixer: {len(mp4_files)} فایل MP4 پیدا شد")

        for src in mp4_files:
            if self._stop_event.is_set():
                log.info("MP4 Fixer: متوقف شد"); return
            self._fix_one(src)

        log.info("MP4 Fixer: تمام فایل‌ها پردازش شدند ✅")

    def _fix_one(self, src):
        dst = src.with_name(src.stem + "_fixed.mp4")
        log.info(f"MP4 Fixer: پردازش {src.name}")

        cmd = [
            FFMPEG_EXE,
            "-i", str(src),

            # ── ویدیو: re-encode با QSV ──
            "-c:v", "h264_qsv",

            # ── فقط این دو تنظیم تغییر می‌کند ──
            "-profile:v", self.profile,
            "-level", self.level,
        ]

        # ── Rate Control ──
        if self.rc_mode == "VBR":
            cmd += ["-b:v", self.bitrate,
                    "-maxrate", self.maxrate,
                    "-bufsize", self.bufsize]
        elif self.rc_mode == "CBR":
            cmd += ["-b:v", self.bitrate,
                    "-minrate", self.bitrate,
                    "-maxrate", self.bitrate,
                    "-bufsize", self.bufsize]
        elif self.rc_mode == "CQP":
            cmd += ["-global_quality", self.cqp]

        # ── بقیه تنظیمات دست‌نخورده ──
        cmd += [
            "-c:a", "copy",              # صدا بدون تغییر
            "-c:s", "copy",              # زیرنویس بدون تغییر (اگر داشته باشه)
            "-map", "0",                 # تمام استریم‌ها
            "-movflags", "+faststart",   # برای پخش سریع‌تر
            "-y", str(dst)
        ]

        log.info("MP4 Fixer CMD: " + " ".join(cmd))

        try:
            result = subprocess.run(
                cmd, check=True,
                capture_output=True,
                creationflags=subprocess.CREATE_NO_WINDOW
            )
            if dst.exists() and dst.stat().st_size > 0:
                log.info(f"MP4 Fixer: ✅ {dst.name} ({dst.stat().st_size // (1024*1024)} MB)")
                if self.delete_src:
                    src.unlink()
                    log.info(f"MP4 Fixer: 🗑️ فایل اصلی حذف شد: {src.name}")
            else:
                log.error(f"MP4 Fixer: ❌ فایل خروجی ساخته نشد!")
        except subprocess.CalledProcessError as e:
            stderr_text = e.stderr.decode(errors='ignore') if e.stderr else "Unknown error"
            log.error(f"MP4 Fixer: ❌ خطای FFmpeg:\n{stderr_text}")

# ════════════════ Guards ════════════════
def priority_guard():
    while True:
        if not throttling_active.is_set(): set_edius_priority()
        time.sleep(10)

def temp_throttle_guard():
    warned = False; throttled = False
    while True:
        t = get_cpu_temp()
        if t and t > TEMP_THRESHOLD:
            if not warned:
                log.warning(f"CPU temp {t:.1f}°C - throttling EDIUS priority")
                warned = True; throttling_active.set()
                pid, handle = get_edius_handle()
                if handle:
                    ctypes.windll.kernel32.SetPriorityClass(handle, NORMAL_PRIORITY_CLASS)
                    ctypes.windll.kernel32.CloseHandle(handle); throttled = True
        else:
            if throttled and t and t <= TEMP_THRESHOLD - 5:
                log.info("CPU cooled – restoring High priority"); throttling_active.clear(); set_edius_priority(); throttled = False
            warned = False
        time.sleep(10)

def memory_guard():
    while True:
        try:
            for flag in (1,2): ctypes.windll.ntdll.NtSetSystemInformation(0x50, ctypes.byref(ctypes.c_ulong(flag)), ctypes.sizeof(ctypes.c_ulong))
        except: pass
        for proc in psutil.process_iter(['pid','name']):
            try:
                if proc.info['name'].lower() in ["system","system idle process",EDIUS_EXE.lower()]: continue
                handle = ctypes.windll.kernel32.OpenProcess(0x1F0FFF, False, proc.info['pid'])
                if handle:
                    ctypes.windll.kernel32.SetProcessWorkingSetSize(handle, -1, -1); ctypes.windll.kernel32.CloseHandle(handle)
            except: continue
        time.sleep(60)

def cleanup():
    log.info("Cleaning up..."); throttling_active.clear(); restore_services()
atexit.register(cleanup)

    def get_advanced_params(self):
        params = {}
        for key, var in self.adv_vars.items():
            if isinstance(var, tk.BooleanVar):
                if var.get(): params[key] = True
            else:
                val = var.get().strip()
                if val and val not in ("none","unknown",""): params[key] = val
        return params

    def _start_mp4_fixer(self):

        maxrate = getattr(self, 'fix_maxrate_var', tk.StringVar(value="20M")).get() if rc == "VBR" else bitrate
        bufsize = getattr(self, 'fix_bufsize_var', tk.StringVar(value="30M")).get() if rc != "CQP" else ""
        cqp     = getattr(self, 'fix_cqp_var', tk.StringVar(value="18")).get() if rc == "CQP" else ""

            folder=folder,

            level=self.fix_level_var.get(),
            rc_mode=rc,
            bitrate=bitrate,
            maxrate=maxrate,
            bufsize=bufsize,
            cqp=cqp,
            delete_src=self.fix_delete_src_var.get()
        )
        self.mp4_fixer.start()

    # ── HQX Converter Start/Stop ──
    def start_converter(self):
        folder = self.watch_folder_var.get()
        if not folder: return
        basic = {
            "video_codec": self.vcodec_var.get(),
            "preset": "veryslow",
            "rate_control": self.rc_var.get(),
            "profile": "high",
            "level": "4.1",
            "gop_size": 250,
            "bframes": 3,
            "look_ahead": True,
            "look_ahead_depth": 60,
            "audio_codec": self.acodec_var.get(),
            "audio_bitrate": self.abitrate_var.get(),
            "delete_source": self.delete_src_var.get(),
        }
        rc = self.rc_var.get()
        if rc == "CQP":
            basic["global_quality"] = int(self.qp_var.get()) if self.qp_var.get().isdigit() else 15
        else:
            basic["bitrate"] = self.bitrate_var.get()
            if rc == "VBR":
                basic["maxrate"] = self.maxrate_var.get()
                basic["bufsize"] = self.bufsize_var.get()
        adv = self.get_advanced_params()
        safe = self.safe_mode_var.get()
        self.converter = HQXConverter(folder, basic, adv, safe); self.converter.start()
        self.start_btn.config(state="disabled"); self.stop_btn.config(state="normal")
        setup_defender_exclusion(folder)

    def stop_converter(self):
        if self.converter: self.converter.stop(); self.converter = None
        self.start_btn.config(state="normal"); self.stop_btn.config(state="disabled")

    def on_close(self):
        if self.converter: self.converter.stop()
        if self.mp4_fixer: self.mp4_fixer.stop()
        if hasattr(self, 'ramdisk_path') and self.ramdisk_path: remove_ramdisk("R:")
        self.overlay.destroy(); self.root.destroy()
